💻 Fast-Path Python Code Example

import rustapi
from pydantic import BaseModel

app = rustapi.Engine()
db = app.connect_db("sqlite::memory:")

class UserCreate(BaseModel):
    name: str
    email: str

@app.post("/users")
def create_user(user: UserCreate):
    db.execute(f"INSERT INTO users (name, email) VALUES ('{user.name}', '{user.email}')")
    return {"status": "created"}

@app.get("/users")
def list_users():
    # Streams query results as JSON directly from sqlx connection pool to Hyper socket
    return db.query_json("SELECT * FROM users")

if __name__ == "__main__":
    app.run(host="127.0.0.1", port=8000)
